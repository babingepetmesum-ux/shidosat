CRUD SEKURITI & SUBSIDI GAS TANGKI PT // ========================================== function saveGuard( bytes32 _companyId, bytes32 _usernameHash, address _guardWallet, bytes calldata _encryptedProfile, bytes32 _initialZoneId, bytes32 _initialPostId, bool _isActive ) external nonReentrant onlyCompanyAdmin(_companyId) { GuardAccount storage g = companyGuards[_companyId][_usernameHash]; if (g.activeWallet == address(0) && _isActive) { companies[_companyId].totalGuards++; } else if (g.isActive && !_isActive && companies[_companyId].totalGuards > 0) { companies[_companyId].totalGuards--; } if (g.activeWallet != address(0)) { delete walletToUsernameHash[_companyId][g.activeWallet]; } g.usernameHash = _usernameHash; g.activeWallet = _guardWallet; g.encryptedProfile = _encryptedProfile; g.currentZoneId = _initialZoneId; g.currentPostId = _initialPostId; g.isActive = _isActive; if (_isActive && _guardWallet != address(0)) { walletToUsernameHash[_companyId][_guardWallet] = _usernameHash; _seedGuardFloat(_companyId, _guardWallet); } emit GuardSaved(_companyId, _usernameHash, _guardWallet, _isActive); } function _seedGuardFloat(bytes32 _companyId, address _guardWallet) internal { Company storage comp = companies[_companyId]; uint256 target = comp.initialGuardFloat; if (_guardWallet.balance < target && target > 0) { uint256 needed = target - _guardWallet.balance; if (comp.gasTankBalance >= needed) { comp.gasTankBalance -= needed; (bool ok, ) = payable(_guardWallet).call{value: needed}(""); if (!ok) comp.gasTankBalance += needed; } } } // ========================================== // 9. KALKULASI HARI LOKAL & DETEKSI 2 / 3 SHIFT (LINTAS TENGAH MALAM) // ========================================== function getCurrentDayIndex(bytes32 _companyId, uint256 _timestamp) public view returns (uint256) { int8 tz = companies[_companyId].shiftRule.timezoneOffset; int256 localSec = int256(_timestamp) + (int256(tz) * 3600); if (localSec < 0) localSec = 0; return uint256(localSec) / 86400; } function _detectShift(bytes32 _companyId, uint256 _timestamp) internal view returns (uint8) { ShiftRule memory r = companies[_companyId].shiftRule; int256 localSec = int256(_timestamp) + (int256(r.timezoneOffset) * 3600); if (localSec < 0) localSec = 0; uint16 minOfDay = uint16((uint256(localSec) % 86400) / 60); if (r.shiftMode == 2) { // Mode 2 Shift (Contoh: Shift 1 Siang 06:00-17:00, Shift 2 Malam 17:00-06:00 lintas tengah malam) if (r.shift1Start < r.shift2Start) { if (minOfDay >= r.shift1Start && minOfDay < r.shift2Start) return 1; return 2; } else { if (minOfDay >= r.shift2Start && minOfDay < r.shift1Start) return 2; return 1; } } else { // Mode 3 Shift (Contoh: Pagi 06:00-14:00, Siang 14:00-22:00, Malam 22:00-06:00) if (minOfDay >= r.shift1Start && minOfDay < r.shift2Start) return 1; if (minOfDay >= r.shift2Start && minOfDay < r.shift3Start) return 2; return 3; } } // ========================================== // 10. SUBMIT PATROLI (ONLINE & OFFLINE BATCH AUTO-SYNC) // ========================================== function _refundGasFromCompanyTank(bytes32 _companyId, uint256 _startGas) internal returns (uint256 refunded) { Company storage comp = companies[_companyId]; uint256 gasUsed = _startGas - gasleft() + GAS_OVERHEAD; uint256 price = tx.gasprice > comp.maxGasPriceRefund ? comp.maxGasPriceRefund : tx.gasprice; uint256 cost = gasUsed * price; if (msg.sender.balance < comp.initialGuardFloat) { uint256 topupNeeded = comp.initialGuardFloat - msg.sender.balance; if (topupNeeded > cost) cost = topupNeeded; } if (comp.gasTankBalance >= cost && cost > 0) { comp.gasTankBalance -= cost; (bool ok, ) = payable(msg.sender).call{value: cost}(""); if (ok) { refunded = cost; } else { comp.gasTankBalance += cost; } } } function _recordPatrolInternal( bytes32 _companyId, bytes32 _uHash, bytes32 _hardwareScannedNfcHash, bytes32 _manualSelectedNfcHash, bool _isPhysicallyTapped, bytes calldata _untappedReasonByGuard, bool _isOfflineSynced, uint256 _verifiedTapTime, uint256 _gasRefunded ) internal { // ANTI-BOHONG (HARDWARE OVERRIDE): // Jika fisik chip NFC di-tap, sistem mengabaikan pilihan manual layar dan mutlak memakai _hardwareScannedNfcHash bytes32 finalNfcHash = _isPhysicallyTapped ? _hardwareScannedNfcHash : _manualSelectedNfcHash; NFCPoint memory pt = companyNFCPoints[_companyId][finalNfcHash]; require(pt.nfcHash != bytes32(0) && pt.isActive, "ShidoGuard: Titik NFC tidak valid/aktif"); // Jika Tertap: Gunakan Keterangan Admin. Jika Tidak Tertap: Gunakan Alasan Sekuriti. bytes memory finalNote = _isPhysicallyTapped ? pt.encryptedDefaultTapNote : _untappedReasonByGuard; GuardAccount storage guard = companyGuards[_companyId][_uHash]; uint256 safeTime = (!_isOfflineSynced || _verifiedTapTime == 0 || _verifiedTapTime > block.timestamp) ? block.timestamp : _verifiedTapTime; require(safeTime >= guard.lastVerifiedTapTime, "ShidoGuard: Manipulasi jam mundur ditolak"); guard.lastVerifiedTapTime = safeTime; // AUTO-ROLLING: Otomatis pindahkan posisi wilayah & pos sekuriti mengikuti NFC yang di-tap if (guard.currentZoneId != pt.zoneId || guard.currentPostId != pt.postId) { guard.currentZoneId = pt.zoneId; guard.currentPostId = pt.postId; emit GuardAutoRolled(_companyId, _uHash, pt.zoneId, pt.postId); } uint256 dayIdx = getCurrentDayIndex(_companyId, safeTime); uint8 shiftIdx = _detectShift(_companyId, safeTime); uint256 logId = companyPatrolLogs[_companyId].length; companyPatrolLogs[_companyId].push(PatrolLog({ logId: logId, dayIndex: dayIdx, usernameHash: _uHash, guardWallet: msg.sender, zoneId: pt.zoneId, postId: pt.postId, nfcHash: finalNfcHash, isPhysicallyTapped: _isPhysicallyTapped, encryptedNote: finalNote, isOfflineSynced: _isOfflineSynced, shiftIndex: shiftIdx, actualTapTime: safeTime, blockRecordedTime: block.timestamp })); nfcDailyStatus[_companyId][finalNfcHash] = NFCDailyStatus({ lastDayIndex: dayIdx, lastTapTime: safeTime, lastGuardHash: _uHash, shiftIndex: shiftIdx, isPhysicallyTapped: _isPhysicallyTapped, encryptedNote: finalNote }); companies[_companyId].totalLogs++; emit PatrolRecorded( _companyId, logId, dayIdx, _uHash, pt.zoneId, pt.postId, finalNfcHash, _isPhysicallyTapped, _isOfflineSynced, shiftIdx, safeTime, _gasRefunded ); } function submitPatrolOnline( bytes32 _companyId, bytes32 _hardwareScannedNfcHash, bytes32 _manualSelectedNfcHash, bool _isPhysicallyTapped, bytes calldata _untappedReasonByGuard ) external nonReentrant onlyActiveGuard(_companyId) { uint256 startGas = gasleft(); bytes32 uHash = walletToUsernameHash[_companyId][msg.sender]; _recordPatrolInternal( _companyId, uHash, _hardwareScannedNfcHash, _manualSelectedNfcHash, _isPhysicallyTapped, _untappedReasonByGuard, false, block.timestamp, 0 ); _refundGasFromCompanyTank(_companyId, startGas); } function submitOfflineBatchPatrol( bytes32 _companyId, bytes32[] calldata _hardwareScannedNfcHashes, bytes32[] calldata _manualSelectedNfcHashes, bool[] calldata _isPhysicallyTappedList, bytes[] calldata _untappedReasons, uint256[] calldata _verifiedTapTimes ) external nonReentrant onlyActiveGuard(_companyId) { uint256 startGas = gasleft(); LAPORAN 1 HARI SEKURITI (HANYA POS SENDIRI & AUTO-RESET HARIAN) // ========================================== function getGuardTodayPostReport( bytes32 _companyId, bytes32 _nfcHash ) external view onlyActiveGuard(_companyId) returns ( bool hasReportToday, bool isPhysicallyTapped, uint8 shiftIndex, uint256 tapTime, bytes memory encryptedNote ) { bytes32 uHash = walletToUsernameHash[_companyId][msg.sender]; GuardAccount memory guard = companyGuards[_companyId][uHash]; NFCPoint memory pt = companyNFCPoints[_companyId][_nfcHash]; require( pt.zoneId == guard.currentZoneId && pt.postId == guard.currentPostId, "ShidoGuard: Anda hanya bisa melihat laporan di Pos bagian Anda sendiri" ); uint256 todayIdx = getCurrentDayIndex(_companyId, block.timestamp); NFCDailyStatus memory st = nfcDailyStatus[_companyId][_nfcHash]; if (st.lastTapTime == 0 || st.lastDayIndex < todayIdx) { return (false, false, 0, 0, ""); } return (true, st.isPhysicallyTapped, st.shiftIndex, st.lastTapTime, st.encryptedNote); } }
