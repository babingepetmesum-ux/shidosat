} } function saveGuard( bytes32 _usernameHash, address _guardWallet, string calldata _profile, bytes32 _initialZoneId, bytes32 _initialPostId, bool _isActive ) external nonReentrant onlyCompanyAdmin { require(_usernameHash != bytes32(0), "ShidoGuard: Hash username kosong");
